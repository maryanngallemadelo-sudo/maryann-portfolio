Product Packaging Analyzer

This Java program stores the number of items packed into 15 boxes using an integer array. It accepts nonnegative integers, calculates the average items per box, identifies the fullest and least-filled boxes, counts boxes containing at least 20 items, and displays the quantities in descending order.


Program Features

  • Accepts quantities for 15 boxes.
  • Validates inputs to prevent negative numbers.
  • Calculates the average items per box.
  • Identifies the fullest box.
  • Identifies the least-filled box.
  • Counts boxes with at least 20 items.
  • Sorts quantities in descending order.

Programming Concepts Used

  • Java Arrays
  • Scanner Class
  • For Loop and While Loop
  • If Statements
  • Conditional Statements
  • Sorting
  • Basic Calculations

Complete Java Source Code


   
import java.util.Scanner;

public class ProductPackagingAnalyzer{
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int[] box = new int[15];
        int[] index = new int[15];

        int total = 0;
        int count = 0;
        int highest = 0;
        int lowest = 0;
        int full = 0;
        int least = 0;

        System.out.println("PRODUCT PACKAGING ANALYZER");

        for (int i = 0; i < 15; i++) {
            index[i] = i;

            System.out.print("Enter items in box " + (i + 1) + ": ");

            while (!sc.hasNextInt()) {
                System.out.print("Invalid! Enter a whole number: ");
                sc.next();
            }

            box[i] = sc.nextInt();

            while (box[i] < 0) {
                System.out.print("Invalid! Enter a nonnegative number: ");

                while (!sc.hasNextInt()) {
                    System.out.print("Invalid! Enter a whole number: ");
                    sc.next();
                }

                box[i] = sc.nextInt();
            }

            total += box[i];

            if (box[i] >= 20) {
                count++;
            }

            if (i == 0 || box[i] > highest) {
                highest = box[i];
                full = i;
            }

            if (i == 0 || box[i] < lowest) {
                lowest = box[i];
                least = i;
            }
        }

        double average = total / 15.0;

        // Sort in descending order
        for (int i = 0; i < 15; i++) {
            for (int j = i + 1; j < 15; j++) {
                if (box[i] < box[j]) {
                    int temp = box[i];
                    box[i] = box[j];
                    box[j] = temp;

                    int tempIndex = index[i];
                    index[i] = index[j];
                    index[j] = tempIndex;
                }
            }
        }

        System.out.println("\n--- RESULTS ---");
        System.out.println("Average: " + average);

        System.out.println("Fullest Box: " + (full + 1));
        System.out.println("Fullest Box Index: " + full);

        System.out.println("Least-filled Box: " + (least + 1));
        System.out.println("Least-filled Box Index: " + least);

        System.out.println("Boxes with at least 20 items: " + count);

        System.out.println("\nDescending Order:");

        for (int i = 0; i < 15; i++) {
            System.out.println("Index " + index[i] + ": " + box[i]);
        }

        sc.close();
    }
}
